  <script>
    // ====================== STATE ======================
    let prices = [];
    let shortSMA = [];
    let longSMA = [];
    let equityCurve = [];
    let trades = [];
    let cash = 10000;
    let position = 0;        // number of shares
    let entryPrice = 0;
    let isRunning = false;
    let intervalId = null;
    let candle = 0;

    // ====================== CHART ======================
    const ctx = document.getElementById('priceChart').getContext('2d');
    const chart = new Chart(ctx, {
      type: 'line',
      data: {
        labels: [],
        datasets: [
          {
            label: 'Price',
            data: [],
            borderColor: '#22d3ee',
            backgroundColor: 'rgba(34, 211, 238, 0.05)',
            borderWidth: 2,
            pointRadius: 0,
            tension: 0.1,
            fill: true
          },
          {
            label: 'Short SMA',
            data: [],
            borderColor: '#22c55e',
            borderWidth: 1.5,
            pointRadius: 0,
            tension: 0.1
          },
          {
            label: 'Long SMA',
            data: [],
            borderColor: '#f97316',
            borderWidth: 1.5,
            pointRadius: 0,
            tension: 0.1
          }
        ]
      },
      options: {
        responsive: true,
        maintainAspectRatio: false,
        interaction: { mode: 'index', intersect: false },
        plugins: {
          legend: {
            labels: { color: '#94a3b8', boxWidth: 12 }
          }
        },
        scales: {
          x: {
            ticks: { color: '#64748b', maxTicksLimit: 10 },
            grid: { color: 'rgba(51,65,85,0.5)' }
          },
          y: {
            ticks: { color: '#64748b' },
            grid: { color: 'rgba(51,65,85,0.5)' }
          }
        }
      }
    });

    // ====================== HELPERS ======================
    function sma(data, period) {
      if (data.length < period) return null;
      const slice = data.slice(-period);
      return slice.reduce((a, b) => a + b, 0) / period;
    }

    function formatMoney(n) {
      return '$' + n.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
    }

    function updateUI() {
      const equity = cash + position * (prices[prices.length - 1] || 100);
      const pnl = equity - parseFloat(document.getElementById('capital').value);
      
      document.getElementById('equity').textContent = formatMoney(equity);
      document.getElementById('pnl').textContent = formatMoney(pnl);
      document.getElementById('pnl').className = `text-xl font-bold mt-1 ${pnl >= 0 ? 'positive' : 'negative'}`;
      document.getElementById('position').textContent = position > 0 ? `LONG ${position}` : 'FLAT';
      document.getElementById('price').textContent = formatMoney(prices[prices.length - 1] || 100);
    }

    function addTradeLog(type, price, shares, pnl = null) {
      const log = document.getElementById('tradeLog');
      if (log.children[0]?.classList.contains('text-slate-500')) log.innerHTML = '';

      const div = document.createElement('div');
      div.className = `p-2 rounded ${type === 'BUY' ? 'signal-buy' : 'signal-sell'}`;
      div.innerHTML = `
        <div class="flex justify-between">
          <span class="font-medium">${type}</span>
          <span>${formatMoney(price)}</span>
        </div>
        <div class="text-xs text-slate-400 mt-0.5">
          ${shares} shares ${pnl !== null ? `• P&L: ${formatMoney(pnl)}` : ''}
        </div>
      `;
      log.prepend(div);
    }

    // ====================== CORE LOGIC ======================
    function nextTick() {
      // Generate next price (random walk)
      const last = prices.length ? prices[prices.length - 1] : 100;
      const change = (Math.random() - 0.48) * 1.8; // slight upward bias
      const newPrice = Math.max(10, last + change);
      prices.push(+newPrice.toFixed(2));

      const shortPeriod = parseInt(document.getElementById('shortSMA').value);
      const longPeriod = parseInt(document.getElementById('longSMA').value);

      const sSMA = sma(prices, shortPeriod);
      const lSMA = sma(prices, longPeriod);

      shortSMA.push(sSMA);
      longSMA.push(lSMA);

      // Trading logic
      if (sSMA !== null && lSMA !== null && prices.length > longPeriod) {
        const prevShort = shortSMA[shortSMA.length - 2];
        const prevLong = longSMA[longSMA.length - 2];

        // Golden Cross → BUY
        if (prevShort <= prevLong && sSMA > lSMA && position === 0) {
          const capital = cash;
          const sizePct = parseFloat(document.getElementById('posSize').value) / 100;
          const invest = capital * sizePct;
          const shares = Math.floor(invest / newPrice);
          if (shares > 0) {
            position = shares;
            entryPrice = newPrice;
            cash -= shares * newPrice;
            addTradeLog('BUY', newPrice, shares);
          }
        }
        // Death Cross → SELL
        else if (prevShort >= prevLong && sSMA < lSMA && position > 0) {
          const pnl = (newPrice - entryPrice) * position;
          cash += position * newPrice;
          addTradeLog('SELL', newPrice, position, pnl);
          position = 0;
          entryPrice = 0;
        }
      }

      // Update chart (keep last 80 candles)
      const maxPoints = 80;
      const start = Math.max(0, prices.length - maxPoints);
      
      chart.data.labels = prices.slice(start).map((_, i) => start + i);
      chart.data.datasets[0].data = prices.slice(start);
      chart.data.datasets[1].data = shortSMA.slice(start);
      chart.data.datasets[2].data = longSMA.slice(start);
      chart.update('none');

      updateUI();
      candle++;
    }

    // ====================== CONTROLS ======================
    document.getElementById('startBtn').addEventListener('click', () => {
      if (isRunning) return;
      isRunning = true;
      document.getElementById('startBtn').disabled = true;
      document.getElementById('pauseBtn').disabled = false;

      // Disable inputs while running
      document.querySelectorAll('input').forEach(i => i.disabled = true);

      const speed = parseInt(document.getElementById('speed').value);
      intervalId = setInterval(nextTick, speed);
    });

    document.getElementById('pauseBtn').addEventListener('click', () => {
      isRunning = false;
      clearInterval(intervalId);
      document.getElementById('startBtn').disabled = false;
      document.getElementById('pauseBtn').disabled = true;
      document.querySelectorAll('input').forEach(i => i.disabled = false);
    });

    document.getElementById('resetBtn').addEventListener('click', () => {
      clearInterval(intervalId);
      isRunning = false;
      prices = [];
      shortSMA = [];
      longSMA = [];
      trades = [];
      cash = parseFloat(document.getElementById('capital').value);
      position = 0;
      entryPrice = 0;
      candle = 0;

      chart.data.labels = [];
      chart.data.datasets.forEach(d => d.data = []);
      chart.update();

      document.getElementById('tradeLog').innerHTML = '<div class="text-slate-500 text-center py-8">No trades yet</div>';
      document.getElementById('startBtn').disabled = false;
      document.getElementById('pauseBtn').disabled = true;
      document.querySelectorAll('input').forEach(i => i.disabled = false);
      updateUI();
    });

    // Initial UI
    updateUI();
  </script>
